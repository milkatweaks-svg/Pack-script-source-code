local function TP_ComputePos(t, tpMethod, tpMinDist, tpMaxDist,
                              dXp, dXm, dYp, dYm, dZp, dZm)
    if tpMethod == "Custom" then return t.Position + Vector3.new(dXp - dXm, dYp - dYm, dZp - dZm) end
    local lo = math.min(tpMinDist, tpMaxDist); local hi = math.max(tpMinDist, tpMaxDist)
    local d = (lo == hi) and lo or (lo + math.random() * (hi - lo))
    local look = (typeof(t) == "Instance" and t.CFrame and t.CFrame.LookVector) or Vector3.new(0, 0, -1)
    if tpMethod == "Behind"      then return t.Position - look * d
    elseif tpMethod == "Ahead"   then return t.Position + look * d
    elseif tpMethod == "Above"   then return t.Position + Vector3.new(0, d, 0)
    elseif tpMethod == "AboveAhead" then return t.Position + look * d + Vector3.new(0, 3, 0)
    elseif tpMethod == "Random"  then local a = math.random() * math.pi * 2; return t.Position + Vector3.new(math.cos(a) * d, 0, math.sin(a) * d)
    elseif tpMethod == "Stutter" then local a = (tick() * 25) % (math.pi * 2); return t.Position + Vector3.new(math.cos(a) * d, 0, math.sin(a) * 0.5)
    end
    return t.Position
end

    return true
end

local voidRandomFlip = true
local function TP_VoidRandomizer()
    local lo, hi = 100000, 500000
    local rx, ry, rz = math.random(lo, hi), math.random(lo, hi), math.random(lo, hi)
    if not voidRandomFlip then rx, ry, rz = -rx, -ry, -rz end
    voidRandomFlip = not voidRandomFlip
    return Vector3.new(rx, ry, rz)
end

local function getHitboxBounds(char)
    local cam = Camera; if not cam or not char then return nil, nil end
    local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge; local found = false
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Transparency < 1 and part.Size.Magnitude > 0 then
            local cf, sz = part.CFrame, part.Size
            local corners = {cf*Vector3.new(-sz.X/2,-sz.Y/2,-sz.Z/2),cf*Vector3.new(sz.X/2,-sz.Y/2,-sz.Z/2),cf*Vector3.new(-sz.X/2,sz.Y/2,-sz.Z/2),cf*Vector3.new(sz.X/2,sz.Y/2,-sz.Z/2),cf*Vector3.new(-sz.X/2,-sz.Y/2,sz.Z/2),cf*Vector3.new(sz.X/2,-sz.Y/2,sz.Z/2),cf*Vector3.new(-sz.X/2,sz.Y/2,sz.Z/2),cf*Vector3.new(sz.X/2,sz.Y/2,sz.Z/2)}
            for _, corner in ipairs(corners) do
                local sp, ok = espWorldToScreen(corner)
                if ok then found = true; minX = math.min(minX, sp.X); minY = math.min(minY, sp.Y); maxX = math.max(maxX, sp.X); maxY = math.max(maxY, sp.Y) end
            end
        end
    end
    if not found then return nil, nil end
    return Vector2.new(math.max(maxX-minX,4), math.max(maxY-minY,8)), Vector2.new(minX, minY)
end

local function drawhealthbar(player, box, hum, size, position, dt)

        if box and box.bars then if box.bars.hp_outline then box.bars.hp_outline.Visible = false end end
        if box and box.bars then for _, line in pairs(box.bars.hp_bars) do if line then line.Visible = false end end end
        return
    end
    local target = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
    local last = box.bars.last_hp or target
    local espCfg = _G.Config.ESP or {}
    local lerpSpeed = espCfg.HealthLerpSpeed or 32
    if espCfg.HighRefresh ~= false then
        dt = dt or (1/(espCfg.TargetHz or 240))
        box.bars.last_hp = last + (target-last)*math.min(1, dt*lerpSpeed)
    else box.bars.last_hp = last + (target-last)*0.05 end
    local lerped = box.bars.last_hp
    local h = math.ceil(size.Y*lerped); local hpx = position.X-5; local hpystart = position.Y+(size.Y-h)
    box.bars.hp_outline.Visible = true
    box.bars.hp_outline.Position = Vector2.new(hpx-1, position.Y-1); box.bars.hp_outline.Size = Vector2.new(3, size.Y+2)

        if i <= segments then
            local segmentheight = h/segments; local yoffset = (i-1)*segmentheight

            line.Color = hpcolor(pospct); line.Thickness = math.max(segmentheight, 1)
        else line.Visible = false end
    end

local function updateesp(player, dt)
    if not player or not ESPObjects[player] then return end
    if not ConfigBox.MasterEnabled then hideesp(player); return end
    if player == LP or checkteammate(player) then hideesp(player); return end
    local char = player.Character
    if not char or espknock(player) then hideesp(player); return end
    local root = char:FindFirstChild("HumanoidRootPart"); local hum = char:FindFirstChildWhichIsA("Humanoid")
    if not root or not hum or hum.Health <= 0 then hideesp(player); return end
    local distance = dist2(player)
    if S.espMaxDist and distance > S.espMaxDist then hideesp(player); return end
    local hrp2d, hrpOk = espWorldToScreen(root.Position)
    if not hrpOk then hideesp(player); return end
    local hitboxSize, hitboxPosition = getHitboxBounds(char)
    local size, position
    if hitboxSize and hitboxPosition then
        size = Vector2.new(math.max(math.floor(hitboxSize.X),4), math.max(math.floor(hitboxSize.Y),8))
        position = Vector2.new(math.floor(hitboxPosition.X), math.floor(hitboxPosition.Y))
    else
        local head = char:FindFirstChild("Head")
        local topWorld = (head and head.Position+Vector3.new(0,0.5,0)) or (root.Position+Vector3.new(0,3,0))
        local bottomWorld = root.Position-Vector3.new(0,3,0)
        local top2d, topOk = espWorldToScreen(topWorld); local bottom2d, bottomOk = espWorldToScreen(bottomWorld)
        if not topOk or not bottomOk then hideesp(player); return end
        local boxHeight = math.abs(bottom2d.Y-top2d.Y); local boxWidth = math.max(boxHeight*0.55, 8)
        size = Vector2.new(math.max(math.floor(boxWidth),4), math.max(math.floor(math.max(boxHeight,10)),8))
        position = Vector2.new(math.floor(hrp2d.X-size.X/2), math.floor(math.min(top2d.Y, bottom2d.Y)))
    end
    local box = ESPObjects[player]
    drawfill(player, box, size, position)
    drawbox(player, box, size, position)
    drawglow(player, box, size, position)
    drawhealthbar(player, box, hum, size, position, dt)
    drawtext(player, box, size, position, distance)
    drawtracers(player, box, root)
end

local function espStepUpdate(dt)
    dt = dt or (1/(_G.Config.ESP and _G.Config.ESP.TargetHz or 240))
    tickval = os.clock()
    if not ConfigBox.MasterEnabled then
        if _espMasterWasOn then
            _espMasterWasOn = false
            for plr in pairs(ESPObjects) do hideesp(plr) end
        end
        return
    end
    _espMasterWasOn = true
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP then
            if not ESPObjects[plr] then CreateBox(plr) end
            updateesp(plr, dt)
        end
    end
end

local function isTeammate(p)
    if not S.teamCheck then return false end
    local a, b = LP:GetAttribute("TeamID"), p:GetAttribute("TeamID")
    return a and b and a ~= "" and a == b
end
local function IsSameTeam(p)
    local a, b = LP:GetAttribute("TeamID"), p:GetAttribute("TeamID")
    return a and b and a ~= "" and a == b
end
local function MyHRP() local c = LP.Character; return c and c:FindFirstChild("HumanoidRootPart") end

local function EnemyList(fn)
    fn = fn or isTeammate; local out = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 and not fn(p) then out[#out+1] = {name=p.Name, hrp=hrp, root=p.Character} end
        end
    end
    return out
end
local function ClosestEnemy()
    local me = MyHRP(); if not me then return nil, nil end
    local best, bestD, bestChar = nil, math.huge, nil
    for _, e in ipairs(EnemyList()) do
        local d = (me.Position - e.hrp.Position).Magnitude
        if d < bestD then bestD = d; best = e.hrp; bestChar = e.root end
    end
    return best, bestChar
end
local function AvgEnemy()
    local list = EnemyList(); if #list == 0 then return nil end
    local sum = Vector3.zero; for _, e in ipairs(list) do sum = sum + e.hrp.Position end; return sum / #list
end
local function EnemyNames()
    local out = {"Closest","Average"}
    for _, e in ipairs(EnemyList(IsSameTeam)) do out[#out+1] = e.name end
    return out
end

local function applyAA()
    if not S.aaEnabled or S.cfDesyncEnabled then return end; local char = LP.Character; if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart"); if not hrp then return end
    local pitch, yaw = 0, 0
    if S.aaPitch == "Flip" then pitch = math.rad(180) elseif S.aaPitch == "Up" then pitch = math.rad(-90) elseif S.aaPitch == "Down" then pitch = math.rad(90) end
    if S.aaSpin and MyHRP() then yaw = math.rad(math.random(-180,180))
    elseif S.aaMode == "Jitter" then S.jitterAngle = (S.jitterAngle + S.aaSpeed * 0.01) % (S.aaYaw * 2); yaw = math.rad(S.jitterAngle - S.aaYaw)
    elseif S.aaMode == "Sway" then yaw = math.sin(S.spinAngle * (S.aaSpeed / 5)) * math.rad(S.aaYaw)
    elseif S.aaMode == "Inverter" then yaw = (math.floor(S.spinAngle * 2) % 2 == 0) and 0 or math.rad(180)
    elseif S.aaMode == "Spin" then yaw = S.spinAngle end
    hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(pitch, yaw, 0)
end
local function aaHB(dt) S.spinAngle = S.spinAngle + (S.aaSpeed * dt); applyAA() end
local function startAA() if S.aaConn then S.aaConn:Disconnect() end; S.spinAngle = 0; S.jitterAngle = 0; S.aaConn = RunService.Heartbeat:Connect(aaHB) end

local crossLines = {}
local function drawCrosshair()
    local c = Camera.ViewportSize / 2; local cs, ct, cg = S.crossSize, S.crossThick, S.crossGap
    local clr = S.crossColor; local spin = S.crossSpin and (tick()*S.crossSpinSpeed*90)%360 or 0
    local cosR, sinR = math.cos(math.rad(spin)), math.sin(math.rad(spin))
    local function rot(v) return Vector2.new(v.X*cosR - v.Y*sinR, v.X*sinR + v.Y*cosR) end
    local lines = {
        {c+rot(Vector2.new(0,cg+cs)), c+rot(Vector2.new(0,cg))},
        {c+rot(Vector2.new(0,-cg-cs)), c+rot(Vector2.new(0,-cg))},
        {c+rot(Vector2.new(cg+cs,0)), c+rot(Vector2.new(cg,0))},
        {c+rot(Vector2.new(-cg-cs,0)), c+rot(Vector2.new(-cg,0))},
    }
    for i=1,4 do
        if not crossLines[i] then local ok,d=pcall(Drawing.new,"Line"); crossLines[i]=ok and d or nil end
    end
    for i=1,4 do local l=crossLines[i]; if l then l.Visible=S.crossEnabled; l.From=lines[i][1]; l.To=lines[i][2]; l.Color=clr; l.Thickness=ct end end
end
local function StartCrosshair() if S.crossConn then S.crossConn:Disconnect() end; S.crossConn=RunService.RenderStepped:Connect(drawCrosshair) end
local function StopCrosshair() if S.crossConn then S.crossConn:Disconnect(); S.crossConn=nil end; for i=1,4 do if crossLines[i] then crossLines[i].Visible=false end end end

local function StartTP()
    if S.tpActive then return end; S.tpActive=true; S.cycLast=tick()
    if (S.tpMethod=="Void Randomizer" or S.tpMethod=="Void Extreme") and S.cycVoid>0 and S.cycEnemy>0 and not S.cycNtf then pcall(function() Library:Notify("Cycle active: alternates between Void Randomizer and Void Extreme") end); S.cycNtf=true end
    S.tpConn = RunService.Heartbeat:Connect(function()
        if not S.tpActive then return end; local me=MyHRP(); if not me then return end; local now=tick()
        if S.autoOpp then S.tpTarget="Closest"; S.orbTarget="Closest"
            if not S.autoOppNtf then pcall(function() Library:Notify("Auto Opponent ON: target=Closest") end); S.autoOppNtf=true end
        end
        local epm; if S.tpMethod=="Void Randomizer" then epm="Void Extreme" elseif S.tpMethod=="Void Extreme" then epm="Void Randomizer" else epm=S.tpMethod end
        local em=S.tpMethod
        if S.cycVoid>0 and S.cycEnemy>0 then
            local pd = (S.cycPhase=="void") and S.cycVoid or S.cycEnemy
            if (now-S.cycLast)>=pd then S.cycLast=now; S.cycPhase=(S.cycPhase=="void") and "enemy" or "void" end
            em=(S.cycPhase=="void") and S.tpMethod or epm
        end
        local isVoid=(em=="Void Randomizer" or em=="Void Extreme")
        if (now-S.tpLast)<(isVoid and 0 or S.tpSpamRate) then return end; S.tpLast=now
        local cf
        if em=="Void Randomizer" then cf=CFrame.new(TP_VoidRandomizer())
        elseif em=="Void Extreme" then cf=CFrame.new(TP_VoidExtreme())
        else local t=tpGetTarget(); if not t then return end

            cf=CFrame.new(pos, (typeof(t)=="Instance") and t.Position or (pos+Vector3.new(0,0,-1)))
        end
        TP_SafeTeleport(me, cf)
    end)
end

local function StartBlink()
    if S.blConn then S.blConn:Disconnect() end
    if MyHRP() then S.blAnchor = MyHRP().Position end
    local lastSnap, anchorT, seed = 0, 0, math.random(1e3, 9e3)
    S.blConn = RunService.Heartbeat:Connect(function(dt)
        if not S.blEnabled or not MyHRP() then return end
        anchorT = anchorT + dt
        S.blAnchor = (S.blAnchor or MyHRP().Position) + Vector3.new(math.sin(anchorT * 1.3 + seed) * 12, math.sin(anchorT * 2.1) * 3, math.cos(anchorT * 1.7 + seed) * 12) * dt
        local now = tick()
        if (now - lastSnap) >= (1 / math.max(S.blRate, 1)) then
            lastSnap = now; local a = math.random() * math.pi * 2; local d = math.random() * S.blRadius
            local hrp = MyHRP()
            if hrp then TP_SafeTeleport(hrp, CFrame.new(S.blAnchor.X + math.cos(a) * d, S.blAnchor.Y + (math.random() - 0.5) * S.blRadius * 0.3, S.blAnchor.Z + math.sin(a) * d)) end
        end
    end)
end

local function screenCenter() return Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2) end
local function GetBestTargetScreenPos()
    local best, bestD = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character and p.Character:FindFirstChild("Head") then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 and not isTeammate(p) then
                local hd = p.Character.Head
                local sp, on = Camera:WorldToViewportPoint(hd.Position)
                if on then
                    local dx = sp.X - Camera.ViewportSize.X * 0.5
                    local dy = sp.Y - Camera.ViewportSize.Y * 0.5
                    local d = dx*dx + dy*dy
                    if d < bestD then bestD = d; best = Vector2.new(sp.X, sp.Y) end
                end
            end
        end
    end
    return best
end

local Lighting = game:GetService("Lighting")
local function StartWeather()
    if S.weatherPart then return end
    local camPart = Instance.new("Part"); camPart.Name = "MehfilWeatherAnchor"
    camPart.Size = Vector3.new(1, 1, 1); camPart.Anchored = true; camPart.CanCollide = false
    camPart.Transparency = 1; camPart.Material = Enum.Material.SmoothPlastic
    camPart.Parent = workspace
    S.weatherPart = camPart

    local function makeEmitter(parent, cfg)
        local e = Instance.new("ParticleEmitter", parent)
        e.Rate = cfg.Rate; e.Speed = cfg.Speed; e.Acceleration = cfg.Acceleration
        e.Lifetime = cfg.Lifetime; e.Size = cfg.Size; e.Color = cfg.Color
        e.Transparency = cfg.Transparency; e.LightEmission = cfg.LightEmission
        e.LightInfluence = 0; e.SpreadAngle = cfg.SpreadAngle
        e.EmissionDirection = cfg.EmissionDirection or Enum.NormalId.Bottom
        e.Orientation = Enum.ParticleOrientation.VelocityParallel
        e.ZOffset = cfg.ZOffset or 0; e.Squash = cfg.Squash or 0
        e.Drag = cfg.Drag or 0; e.RotSpeed = cfg.RotSpeed or NumberRange.new(0)
        return e
    end
    local att = Instance.new("Attachment", camPart); att.Name = "WeatherAttach"
    if S.weatherType == "Rain" then

        S.weatherEmitter = makeEmitter(att, {
            Rate = S.weatherDensity * 3,
            Speed = NumberRange.new(S.weatherSpeed * 80, S.weatherSpeed * 120),
            Acceleration = Vector3.new(0, -300, 0),
            Lifetime = NumberRange.new(0.5, 1.2),
            Size = NumberSequence.new(S.weatherSize * 0.04, S.weatherSize * 0.02),
            Color = ColorSequence.new(S.weatherColor),
            Transparency = NumberSequence.new(0.4, 0.9),
            LightEmission = 0.15,
            SpreadAngle = Vector2.new(2, 2),
            EmissionDirection = Enum.NormalId.Bottom,
            ZOffset = 0,
            Squash = -0.85,
            Drag = 0.1,
            RotSpeed = NumberRange.new(-20, 20)
        })
    else

        S.weatherEmitter = makeEmitter(att, {
            Rate = S.weatherDensity,
            Speed = NumberRange.new(S.weatherSpeed * 4, S.weatherSpeed * 12),
            Acceleration = Vector3.new(math.random(-8, 8), -20, math.random(-8, 8)),
            Lifetime = NumberRange.new(3, 7),
            Size = NumberSequence.new(S.weatherSize * 0.6, S.weatherSize * 0.3),
            Color = ColorSequence.new(S.weatherColor),
            Transparency = NumberSequence.new(0.1, 0.6),
            LightEmission = 0.3,
            SpreadAngle = Vector2.new(50, 50),
            EmissionDirection = Enum.NormalId.Bottom,
            ZOffset = 20,
            Squash = 0,
            Drag = 0.5,
            RotSpeed = NumberRange.new(-60, 60)
        })
    end
    S.weatherEmitter.Enabled = true

    S.weatherConn = RunService.RenderStepped:Connect(function()
        if not S.weatherEnabled or not S.weatherPart or not Camera then return end
        local cf = Camera.CFrame
        S.weatherPart.CFrame = CFrame.new(cf.Position + cf.LookVector * 8 + Vector3.new(0, 18, 0))
    end)

    local sid = S.weatherSounds[S.weatherType]
    if sid and S.weatherVolume > 0 then
        local snd = Instance.new("Sound"); snd.SoundId = sid; snd.Volume = S.weatherVolume
        snd.Looped = true; snd.Parent = workspace; snd:Play(); S.weatherSound = snd
    end
end
local function StopWeather()
    if S.weatherConn then S.weatherConn:Disconnect(); S.weatherConn = nil end
    if S.weatherSound then S.weatherSound:Stop(); S.weatherSound:Destroy(); S.weatherSound = nil end
    if S.weatherEmitter then S.weatherEmitter:Destroy(); S.weatherEmitter = nil end
    if S.weatherPart then S.weatherPart:Destroy(); S.weatherPart = nil end
end
local function RefreshWeather()
    if not S.weatherEnabled then StopWeather(); return end
    StopWeather(); task.wait(0.05); S.weatherPart = nil; S.weatherEmitter = nil; StartWeather()
end

local function EnsurePostEffect(name, className)
    local existing = Lighting:FindFirstChild(name)
    if existing then return existing end
    local fx = Instance.new(className); fx.Name = name; fx.Enabled = false; fx.Parent = Lighting
    return fx
end
local function ApplyPostProcessing()
    if not (S.mbEnabled or S.bloomEnabled or S.srEnabled or S.ccEnabled or S.fogEnabled or S.customTime or S.atmoEnabled or S.shadowSoftness ~= 0.5) then return end

    local blur = EnsurePostEffect("MehfilBlur", "BlurEffect")
    blur.Enabled = S.mbEnabled; blur.Size = S.mbIntensity * 56

    local bloom = EnsurePostEffect("MehfilBloom", "BloomEffect")
    bloom.Enabled = S.bloomEnabled; bloom.Intensity = S.bloomIntensity
    bloom.Size = S.bloomSize; bloom.Threshold = S.bloomThreshold

    local sr = EnsurePostEffect("MehfilSunRays", "SunRaysEffect")
    sr.Enabled = S.srEnabled; sr.Intensity = S.srIntensity; sr.Spread = S.srSpread

    local cc = EnsurePostEffect("MehfilCC", "ColorCorrectionEffect")
    cc.Enabled = S.ccEnabled; cc.Brightness = S.ccBrightness; cc.Contrast = S.ccContrast
    cc.Saturation = S.ccSaturation; cc.TintColor = S.ccTintColor

    if S.fogEnabled then
        Lighting.FogColor = S.fogColor; Lighting.FogStart = S.fogStart; Lighting.FogEnd = S.fogEnd
    else Lighting.FogEnd = 100000 end

    if S.shadowSoftness ~= 0.5 then
        pcall(function() Lighting.GlobalShadows = true end)
        Lighting.EnvironmentDiffuseScale = 1 - (S.shadowSoftness * 0.5)
        Lighting.EnvironmentSpecularScale = 1 - (S.shadowSoftness * 0.3)
    end

    if S.customTime then
        Lighting.ClockTime = S.timeHour; Lighting.Brightness = S.customBrightness
    end

    local atmo = Lighting:FindFirstChildOfClass("Atmosphere")
    if S.atmoEnabled then
        if not atmo then atmo = Instance.new("Atmosphere"); atmo.Parent = Lighting end
        atmo.Density = S.atmoDensity; atmo.Offset = S.atmoOffset; atmo.Haze = S.atmoHaze
        atmo.Glare = S.atmoGlare; atmo.Color = S.atmoColor; atmo.Decay = S.atmoDecay
    elseif atmo and not S.atmoEnabled then
        pcall(function() atmo:Destroy() end)
    end
end

local function loadConfigAsync()
    if not isfile or not isfile("config.json") then return warn("[Mehfil] No config.json found") end
    local ok, raw = pcall(readfile, "config.json")
    if not ok then return warn("[Mehfil] Failed to read config.json") end
    local HttpService = game:GetService("HttpService")
    local ok2, data = pcall(HttpService.JSONDecode, HttpService, raw)
    if not ok2 or not data or not data.objects then return warn("[Mehfil] Invalid config.json") end
    local objects = data.objects
    local total = #objects

    local BATCH = 12
    local count, skipped = 0, 0
    local function applyOne(key, ctrl)
        if not ctrl then return end
        pcall(function() ctrl:SetValue(key) end)
    end
    for i = 1, total, BATCH do
        local batchEnd = math.min(i + BATCH - 1, total)
        for j = i, batchEnd do
            local obj = objects[j]
            local idx, t, val = obj.idx, obj.type, obj.value
            local ctrl = Toggles[idx] or Options[idx]
            if ctrl then
                if t == "Toggle" then
                    applyOne(val == true or val == "true", ctrl)
                elseif t == "Slider" then
                    local n = tonumber(val)
                    if n then applyOne(n, ctrl) end
                elseif t == "Dropdown" then
                    applyOne(tostring(val), ctrl)
                elseif t == "ColorPicker" then
                    local hex = tostring(val or "ffffff")
                    local r = tonumber(hex:sub(1,2), 16)
                    local g = tonumber(hex:sub(3,4), 16)
                    local b = tonumber(hex:sub(5,6), 16)
                    if r and g and b then applyOne(Color3.fromRGB(r, g, b), ctrl) end
                end
                count = count + 1
            else
                skipped = skipped + 1
            end
        end

        if batchEnd < total then task.wait() end
    end
    warn(string.format("[Mehfil] Config loaded: %d applied, %d skipped, %d total", count, skipped, total))
end
task.spawn(function() task.wait(0.5); loadConfigAsync() end)
